    T = owner():lastSaid():explode(" ")

        ManualOverRide = 1

        print("COMMAND-SENT!")
    }
        
    # Opposite of MOR; "Ease Of Controls" (EOC)
        
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "EOC") {
        ManualOverRide = 0
        hideChat(1)
        print("COMMAND-SENT!")
    }
