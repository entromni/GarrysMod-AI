if(owner():steamID()!= "STEAM_0:0:441199989"){
        selfDestruct()

## Constants/Variables:

### All the following are security commands... Have fun and stay safe... ###

runOnChat(1)

if(chatClk(owner())) {
    
    # Manual Over Ride (MOR)
    
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "MOR") {
        ACTIVATION = 1
        print("COMMAND-SENT!")
    }
        
    # Opposite of MOR; "Ease Of Controls" (EOC)
        
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "EOC") {
        ACTIVATION = 0
        print("COMMAND-SENT!")
    }
        
    if(ACTIVATION==1) {
        if(chatClk(owner())) {
                
            # Forces the target to be owner.
        
            T = owner():lastSaid():explode(" ")
            if(T[1, string] == "FOLLOW") {
                Target = owner()
                print("COMMAND-SENT!")
            }

        }
    }
}

## Throttle:

    Throttle = 50

if(TURNr == -1 & TURNl == 1) {
    Forwardrw = -1, Forwardlw = 1
}

    Forwardrw = 1, Forwardlw = -1
