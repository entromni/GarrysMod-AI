    T = owner():lastSaid():explode(" ")

        Power = 1, Encode = round(random(1,10))
        hideChat(1)
        print("Drone Successfully Activate"), print("To deactivate this drone type 'PowerDown' followed by this code: "+toString(Encode))
    }
    
}
    
    if(chatClk(owner())) {
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "PowerDown"+toString(Encode) ) {
        Power = 0
        hideChat(1)
        print("Drone Successfully Powered Down")
    }
}
    
    # Manual Over Ride (MOR)
    
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "MOR") {
        ManualOverRide = 1
        hideChat(1)
        print("COMMAND-SENT!")
    }
        
    # Opposite of MOR; "Ease Of Controls" (EOC)
        
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "EOC") {
        ManualOverRide = 0
        hideChat(1)
        print("COMMAND-SENT!")
    }
        
    if(ManualOverRide==1) {
        if(chatClk(owner())) {
                
            # Forces the target to be owner.
        
            T = owner():lastSaid():explode(" ")
            if(T[1, string] == "FOLLOW") {
                Target = owner()
                hideChat(1)
                print("COMMAND-SENT!")
            }

        }
    }
}
