@name Defense drone experimental 3
@inputs [TTVr TTVl]:number [Insight Target]:entity [Distance]:number Base:entity 
@outputs [LEFT FRONT BACK RIGHT]:number [TURNr TURNl]:number [Forwardrw Forwardlw Backwardrw Backwardlw]:number [Speedmod]:number [RIGHTo LEFTo]:number [TurretFire Aim]:number [TimerDisplayOut RandomizerDisplayOut]:number Throttle
@persist [PATROL HUNT RANDOMIZE TIMERTURNING]:number ACTIVATION:number 

@persist [RangerLeft RangerRight RangerBack RangerFront]:ranger
@persist [RangerLeftHullSize RangerRightHullSize RangerFrontHullSize RangerBackHullSize]:vector
@persist [RangerLeftOffset RangerRightOffset RangerFrontOffset RangerBackOffset]:number
@persist ShowRaycastHolos
@persist FilterArray:array Length Frequency FilterProps

### All the following are security commands... Have fun and stay safe... ###

if(first()|dupefinished()){
    
    runOnTick(1)
        
        ShowRaycastHolos = 1
        
            RangerLeftOffset = 0
            RangerRightOffset = 180
            RangerFrontOffset = 90
            RangerBackOffset = 270
            
            RangerLeftHullSize = vec(1,1,1)
            RangerRightHullSize = vec(1,1,1)
            RangerFrontHullSize = vec(1,1,1)
            RangerBackHullSize = vec(1,1,1)

        FilterProps = 1

Frequency = 10
Length = 400

}

    LEFT = RangerLeft:distance()
    RIGHT = RangerRight:distance()
    FRONT = RangerFront:distance()
    BACK = RangerBack:distance()

## The following is for turning:

if(FRONT < 300 & RIGHT < 300) {
    Forwardrw = 1, Forwardlw = -1, Backwardrw = -1, Backwardlw = 1
}

elseif(FRONT < 300 & LEFT < 300) {
    Forwardrw = -1, Forwardlw = 1, Backwardrw = 1, Backwardlw = -1
}

elseif(RIGHT < 300) {
    Forwardrw = -1, Forwardlw = 1, Backwardrw = 1, Backwardlw = -1
}

elseif(LEFT < 300) {
    Forwardrw = -1, Forwardlw = 1, Backwardrw = 1, Backwardlw = -1
}

elseif(FRONT < 300 & RIGHT > 100 & LEFT > 200) {
    Forwardrw = -1, Forwardlw = 1, Backwardrw = 1, Backwardlw = -1
}

if(FRONT < 300 & LEFT > 100 & RIGHT > 200) {
    Forwardrw = 1, Forwardlw = -1, Backwardrw = -1, Backwardlw = 1
}

else { 
    TURNl = 0, TURNr = 0
}

## The following is for back & forward movement:

# Front:

if(FRONT < 300) {
    Backwardrw = -1, Backwardlw = -1, Forwardrw = -1, Forwardlw = -1
}

elseif(FRONT > 300) {
    Backwardrw = 1, Backwardlw = 1, Forwardrw = 1, Forwardlw = 1
}

elseif(BACK < 100) {
    Backwardrw = 1, Backwardlw = 1, Forwardrw = 1, Forwardlw = 1
}

else {
    Backwardrw = 0, Backwardlw = 0, Forwardrw = 0, Forwardlw = 0
}

timer("RaycastA", Frequency)

if(clk("RaycastA")) 

{

### Raycasts: ####---------------------------------------------------------------------------####

## Main:

    rangerFilter(FilterArray)
        RangerLeft = rangerAngleHull( Length, entity():toLocal(Base:angles()):yaw()+RangerLeftOffset, 90, RangerLeftHullSize)
    rangerFilter(FilterArray)
        RangerRight = rangerAngleHull( Length, entity():toLocal(Base:angles()):yaw()+RangerRightOffset, 90, RangerRightHullSize)
    rangerFilter(FilterArray)
        RangerFront = rangerAngleHull( Length, entity():toLocal(Base:angles()):yaw()+RangerFrontOffset, 90, RangerFrontHullSize)
    rangerFilter(FilterArray)
        RangerBack = rangerAngleHull( Length, entity():toLocal(Base:angles()):yaw()+RangerBackOffset, 90, RangerBackHullSize)

        if(ShowRaycastHolos == 1)

        {

            holoCreate(1)
            holoCreate(2)
            holoCreate(3)
            holoCreate(4)
    
            holoPos(1, RangerLeft:position())
            holoPos(2, RangerRight:position())
            holoPos(3, RangerFront:position())
            holoPos(4, RangerBack:position())
    
            holoScale(1, RangerLeftHullSize)
            holoScale(2, RangerRightHullSize)
            holoScale(3, RangerFrontHullSize)
            holoScale(4, RangerBackHullSize)
    
            holoColor(1, vec(0,0,255))
            holoColor(2, vec(0,255,0))
            holoColor(3, vec(255,0,0))    
            holoColor(4, vec(255,255,255))  

        }


}
    
if(FilterProps == 1) 

{
    if(RangerLeft:entity():owner()==owner()){FilterArray[FilterArray:count()+1,entity]=RangerLeft:entity()}
    if(RangerRight:entity():owner()==owner()){FilterArray[FilterArray:count()+1,entity]=RangerRight:entity()}
    if(RangerFront:entity():owner()==owner()){FilterArray[FilterArray:count()+1,entity]=RangerFront:entity()}
    if(RangerBack:entity():owner()==owner()){FilterArray[FilterArray:count()+1,entity]=RangerBack:entity()}

    timer("RaycastB", 2000)

}

if(clk("RaycastB")) 

{

    FilterProps = 0

}
