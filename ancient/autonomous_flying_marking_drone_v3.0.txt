event chat(Ply:entity, Said:string, _:number) {

    T = Said:explode(" ")

    if(Ply==owner() ? 1 : 0 ) { 
    if(T[1, string] == CommandPrefix+"Activate"+toString(AI_Identification) ) {
        Power = 1, AI_Identification = round(random(1,10))
        hideChat(1)
        print("Robot successfully activated. To deactivate this robot type 'PowerDown' followed by this code: "+toString(AI_Identification))
    }

}

    if(Ply==owner() ? 1 : 0 ) { 
    if(T[1, string] == CommandPrefix+"ActivateAll" ) {
        Power = 1
        hideChat(1)
        print("All robots successfully activated")
    }
    
}

    if(Ply==owner() ? 1 : 0 ) { 
    if(T[1, string] == CommandPrefix+"PowerDown"+toString(AI_Identification) ) {
        Power = 0, AI_Identification = round(random(1,10))
        hideChat(1)
        print("Robot successfully power-down. To activate this robot type 'Activate' followed by this code: "+toString(AI_Identification))
    }
}

    if(Ply==owner() ? 1 : 0 ) { 
    if(T[1, string] == CommandPrefix+"DeactivateAll" ) {
        Power = 0
        hideChat(1)
        print("All robots successfully deactivated")
    }

}

    if(Power == 1) 

    {

    # Manual Over Ride (MOR)
    
    if(Ply==owner() ? 1 : 0 ) { 
    if(T[1, string] == CommandPrefix+"MOR"+toString(AI_Identification)) {
        ManualOverRide = 1
        hideChat(1)
        print("Manual over-ride activated")
    } 
    
}

    if(Ply==owner() ? 1 : 0 ) { 
    if(T[1, string] == CommandPrefix+"SetWayPoint") {
        WayPoint_Exists = 1
        
        WayPoint_List:pushVector(Ply:pos())
        WayPoint = WayPoint_List[1, vector]
        
        hideChat(1)
        print("Set waypoint to: "+WayPoint:toString())
    } 
    
}

    # Manual Over Ride All (MORA)
    
    if(Ply==owner() ? 1 : 0 ) { 
    if(T[1, string] == CommandPrefix+"MORA") {
        ManualOverRide = 1
        hideChat(1)
        print("Manual over-rided all robots")
    }
}

    # Opposite of MOR: "Ease Of Controls" (EOC)
        
    if(Ply==owner() ? 1 : 0 ) { 
    if(T[1, string] == CommandPrefix+"EOC"+toString(AI_Identification)) {
        ManualOverRide = 0, Follow = 0
        hideChat(1)
        print("Manual over-ride de-activated")

    }

}

}
        
    if(ManualOverRide == 1) {
        if(Ply==owner() ? 1 : 0 ) { 
                
            # Forces a singular robot follow you.
            if(T[1, string] == CommandPrefix+"Follow"+toString(AI_Identification)) {
                Follow = 1
                hideChat(1)
                print("Robot is now following you...")

                }

            }
            
        if(Ply==owner() ? 1 : 0 ) { 
            
            # Makes all robots follow you.
            if(T[1, string] == CommandPrefix+"FollowAll") {
                Follow = 1
                hideChat(1)
                print("All robots are now following you...")

                }

            }
            
        if(Ply==owner() ? 1 : 0 ) { 
            
            # Stops all robots from following.
            if(T[1, string] == CommandPrefix+"FollowAll") {
                Follow = 1
                hideChat(1)
                print("All robots are now following you...")

                }

            }

        if(Ply==owner() ? 1 : 0 ) { 
                
            # Disables targeting.
            if(T[1, string] == CommandPrefix+"TargetingOff"+toString(AI_Identification)) {
                Targeting = 0
                hideChat(1)
                print("Successfully disabled targeting")

                }

            }

        if(Ply==owner() ? 1 : 0 ) { 
                
            # Disables movement.
            if(T[1, string] == CommandPrefix+"MovementOff"+toString(AI_Identification)) {
                Movement = 0
                hideChat(1)
                print("Successfully disabled movement")

              }

           }

        }

}
