function number perceptron(X:array, Output:number, Predict:number){
        OutputP = 0
        if (Predict == 1){
            foreach(I, Value:number=X){
                OutputP += Value*Weights[I, number]
            }
            OutputP += Bias*Weights[Weights:count(), number]
            return OutputP
        } else {
            foreach(I, Value:number=X){
                OutputP += Value*Weights[I, number]
            }
            OutputP += Bias*Weights[Weights:count(), number]
            Error = Output - OutputP
            foreach(I, Value:number=X){
                Weights[I, number] = Weights[I, number] + (Lr*Error*Value)
            }
            Weights[Weights:count(), number] = Weights[Weights:count(), number] + (Lr*Error*Bias)
            return Error
        }
}
