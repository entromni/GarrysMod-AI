        print("Manual over-ride activated")
    }
        
    # Opposite of MOR: "Ease Of Controls" (EOC)
        
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "!EOC"+toString(Encode)) {
        ManualOverRide = 0, Follow = 0
        hideChat(1)
        print("Manual over-ride de-activated")

}
        
    if(ManualOverRide == 1) {
        if(chatClk(owner())) {
                
            # Forces the target to be owner.
        
            T = owner():lastSaid():explode(" ")
            if(T[1, string] == "!Follow"+toString(Encode)) {
                Follow = 1
                hideChat(1)
                print("Robot is now following you...")

                }

            LGoto = Check:position() + Check:hitNormal() * MGround
            
            local Dir = LGoto - LHold
            if (Dir:length() > MWidth * MStep) {
                local Mid = mix(LGoto, LHold, 0.85) + Dir:toAngle():up() * Dir:length() * MStep
                LPos = mix(bezier(LHold, Mid, LGoto, LTime), LPos, LTime)
                LOsc = cos(360 * LTime) * MHeight * -MOsc
            }
            else {
                LOsc = 0    
            }
        }
    }
}
else {

        LPos = LGoto
        LMove = 0    
    }
}
