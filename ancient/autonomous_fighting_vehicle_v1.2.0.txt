#### Chat commands: ####------------------------------------------------------------------------####

runOnChat(1)

if(chatClk(owner())) {
    
    if(chatClk(owner())) {
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "Activate"+toString(Encode) ) {
        Power = 1, Encode = round(random(1,10))
        hideChat(1)
        print("Robot successfully activated"), print("To deactivate this robot type 'PowerDown' followed by this code: "+toString(Encode))
    }

}

    if(chatClk(owner())) {
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "ActivateAll" ) {
        Power = 1
        hideChat(1)
        print("All robots successfully activated")
    }
    
}

    if(chatClk(owner())) {
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "PowerDown"+toString(Encode) ) {
        Power = 0
        hideChat(1)
        print("Robot successfully powered-down")
    }
}

    if(chatClk(owner())) {
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "DeactivateAll" ) {
        Power = 0
        hideChat(1)
        print("All robots successfully deactivated")
    }

}

    if(Power == 1) 

    {

    # Manual Over Ride (MOR)
    
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "MOR"+toString(Encode)) {
        ManualOverRide = 1
        hideChat(1)
        print("Manual over-ride activated")
    }
        
    # Opposite of MOR: "Ease Of Controls" (EOC)
        
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "EOC"+toString(Encode)) {
        ManualOverRide = 0, FOLLOW = 0
        hideChat(1)
        print("Manual over-ride de-activated")
    }
        
    if(ManualOverRide == 1) {
        if(chatClk(owner())) {
                
            # Forces the target to be owner.
        
            T = owner():lastSaid():explode(" ")
            if(T[1, string] == "!Follow"+toString(Encode)) {
                FOLLOW = 1
                hideChat(1)
                print("Robot is now following you...")

                }

            }

        if(chatClk(owner())) {
                
            # Disables targeting.
        
            T = owner():lastSaid():explode(" ")
            if(T[1, string] == "Disable Targeting"+toString(Encode)) {
                Targeting = 0
                hideChat(1)
                print("Successfully disabled targeting")

                }

            }
        if(chatClk(owner())) {
                
            # Disables movement.
        
            T = owner():lastSaid():explode(" ")
            if(T[1, string] == "Disable Movement"+toString(Encode)) {
                Movement = 0
                hideChat(1)
                print("Successfully disabled movement")

                }

            }

        }

    }

}

if(Power == 0) { TurretFire = 0, LatchTurret = 1, stopAllTimers() }

    HuntY = Base:bearing(Target:pos())

    Distance = Base:pos():distance(Target:pos())

    }

## Smoke launching:

if(Distance < 8000 & HUNT == 1) {

    timer("Smoke", 2000)

    if(clk("Smoke")) {

    RandomSmoke = round(random(-3,3))

    stoptimer("Smoke")

    }

    if(RandomSmoke == 3) { SmokeLaunch = 1, timer("Smoke", 100) } elseif(RandomSmoke != 3) { SmokeLaunch = 0, timer("Smoke", 1000) }

}

else { stoptimer("Smoke") }

## Turret aiming:

if(Aim == 1|Follow == 0 & Protect != 0) {

LatchTurret = 0

if(Target) 

{ 

timer("Aim", 100)

if(clk("Aim"))

{

if(Insight) { TurretFire = 1 } else { TurretFire = 0 }

if(Distance > 3000 ) { DistanceTargetPitchOffset = 1 } elseif(Distance < 3000 & Distance > 250 ) { DistanceTargetPitchOffset = 0.1 } else { DistanceTargetPitchOffset = 0 }

holoCreate(5, Haxis:pos(), vec(0.3), Self:toWorld(ang(0, TurretAng:yaw(), 0)))

holoParent(5, Self)

holoCreate(6, Vaxis:pos(), vec(0.3), Turret:angles():setRoll(0))  

holoParent(6, 5)

Vaxis:parentTo(holoEntity(6))
Haxis:parentTo(holoEntity(5))
Turret:parentTo(holoEntity(6))

TurretAng = Self:toLocal(Turret:angles())

    holoAng(5, HaxisAngle)
    holoAng(6, VaxisAngle)

    HaxisYaw = (Base:pos() - Target:pos()):toAngle():yaw() - 270
    HaxisPitch = Base:angles():pitch()
    HaxisRoll  = Base:angles():roll()

    HaxisAngle = ang(HaxisRoll, HaxisYaw, HaxisPitch)

    VaxisYaw = Haxis:angles():yaw() - 90
    VaxisPitch = Base:angles():pitch() - (Base:pos() - Target:pos()):toAngle():pitch() + DistanceTargetPitchOffset
    VaxisRoll  = Base:angles():roll()

    VaxisAngle = ang(VaxisPitch, VaxisYaw, VaxisRoll)

holoAlpha(5, 0)
holoAlpha(6, 0)

        }

    }

}

else { TurretFire = 0, LatchTurret = 1, stoptimer("Aim") }

}

    rangerFilter(FilterArray)

    rangerFilter(FilterArray)

        RangerFront = rangerAngleHull( Length, entity():toLocal(Base:angles()):yaw()+RangerFrontOffset, 90, RangerFrontHullSize)
    rangerFilter(FilterArray)
        RangerBack = rangerAngleHull( Length, entity():toLocal(Base:angles()):yaw()+RangerBackOffset, 90, RangerBackHullSize)

        if(ShowRaycastHolos == 1)

        {

            holoCreate(1)
            holoCreate(2)
            holoCreate(3)
            holoCreate(4)

            holoPos(1, RangerLeft:position())
            holoPos(2, RangerRight:position())
            holoPos(3, RangerFront:position())
            holoPos(4, RangerBack:position())
    
            holoScale(1, RangerLeftHullSize)
            holoScale(2, RangerRightHullSize)
            holoScale(3, RangerFrontHullSize)
            holoScale(4, RangerBackHullSize)
    
            holoColor(1, vec(0,0,255))
            holoColor(2, vec(0,255,0))
            holoColor(3, vec(255,0,0))    
            holoColor(4, vec(255,255,255))  

        }

}

elseif(Distance > 3000 & TURN == 0 & HUNT == 1 & sign(HuntY) == -1) 

{ RightTransfer["Gear",number] = 1, LeftTransfer["Gear",number] = 1

RightTransfer["Left Brake",number] = 50, RightTransfer["Right Brake",number] = 50, RightTransfer["Left Clutch",number] = 1, RightTransfer["Right Clutch",number] = 1
LeftTransfer["Left Brake",number] = 0, LeftTransfer["Right Brake",number] = 0, LeftTransfer["Left Clutch",number] = 0, LeftTransfer["Right Clutch",number] = 0

}

elseif(Distance < 3000 & Distance > 250 & TURN == 0 & HUNT == 1 & sign(HuntY) == 1) { RightTransfer["Gear",number] = 1, LeftTransfer["Gear",number] = 2 }
elseif(Distance < 3000 & Distance > 250 & TURN == 0 & HUNT == 1 & sign(HuntY) == -1) { RightTransfer["Gear",number] = 2, LeftTransfer["Gear",number] = 1 }

elseif(FRONT < 700 & RIGHT < 700) { TURN = 1, RightTransfer["Gear",number] = 2, LeftTransfer["Gear",number] = 1 }
elseif(FRONT < 700 & LEFT < 700) { TURN = 1, RightTransfer["Gear",number] = 1, LeftTransfer["Gear",number] = 2 }

elseif(RIGHT > 100 & LEFT > 100 & RIGHT < 800 & LEFT < 800) { TURN = 1, RightTransfer["Gear",number] = 1, LeftTransfer["Gear",number] = 1 }

elseif(RIGHT < 700) { TURN = 1, RightTransfer["Gear",number] = 2, LeftTransfer["Gear",number] = 1 }
elseif(LEFT < 700) { TURN = 1, RightTransfer["Gear",number] = 1, LeftTransfer["Gear",number] = 2 }
