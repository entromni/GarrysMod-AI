#### Chat commands: ####------------------------------------------------------------------------####

runOnChat(1)

if(chatClk(owner())) {

    if(chatClk(owner())) {
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "Activate"+toString(Encode) ) {
        Power = 1, Encode = round(random(1,10))
        hideChat(1)
        print("Robot successfully activated. To deactivate this robot type 'PowerDown' followed by this code: "+toString(Encode))
    }

}

    if(chatClk(owner())) {
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "!ActivateAll" ) {
        Power = 1
        hideChat(1)
        print("All robots successfully activated")
    }
    
}

    if(chatClk(owner())) {
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "PowerDown"+toString(Encode) ) {
        Power = 0, Encode = round(random(1,10))
        hideChat(1)
        print("Robot successfully power-down. To activate this robot type 'Activate' followed by this code: "+toString(Encode))
    }
}

    if(chatClk(owner())) {
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "!DeactivateAll" ) {
        Power = 0
        hideChat(1)
        print("All robots successfully deactivated")
    }

}

    if(chatClk(owner())) {
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "NetworkClone"+toString(Encode) ) {
    local ChosenNetwork = Networks:clone()
    ChosenNetwork:nRandomizeWeights(table("range" = array(-MutationRate, MutationRate)))
    Networks[1, table] = ChosenNetwork
    Generation++
    print(_HUD_PRINTCENTER, "Generation: " + Generation)
        hideChat(1)
        print("Network cloned")
    }
}

    if(Power == 1) 

    {

    # Manual Over Ride (MOR)
    
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "!MOR"+toString(Encode)) {
        ManualOverRide = 1
        hideChat(1)
        print("Manual over-ride activated")
    }
        
    # Opposite of MOR: "Ease Of Controls" (EOC)
        
    T = owner():lastSaid():explode(" ")
    if(T[1, string] == "!EOC"+toString(Encode)) {
        ManualOverRide = 0, Follow = 0
        hideChat(1)
        print("Manual over-ride de-activated")

    }
    
}
        
    if(ManualOverRide == 1) {
        if(chatClk(owner())) {
                
            # Forces the target to be owner.
        
            T = owner():lastSaid():explode(" ")
            if(T[1, string] == "!Follow"+toString(Encode)) {
                Follow = 1
                hideChat(1)
                print("Robot is now following you...")

                }

            }

        if(chatClk(owner())) {
                
            # Disables targeting.
        
            T = owner():lastSaid():explode(" ")
            if(T[1, string] == "!TargetingOff"+toString(Encode)) {
                Targeting = 0
                hideChat(1)
                print("Successfully disabled targeting")

                }

            }

        if(chatClk(owner())) {
                
            # Disables movement.
        
            T = owner():lastSaid():explode(" ")
            if(T[1, string] == "!MovementOff"+toString(Encode)) {
                Movement = 0
                hideChat(1)
                print("Successfully disabled movement")

            }

        }

    }

}

## Turret aiming:

if(Target) 

{

{

if(Distance > 3000 ) { DistanceTargetPitchOffset = 0 } elseif(Distance < 3000 & Distance > 1000 ) { DistanceTargetPitchOffset = 0.5 } elseif(Distance < 1000 & Distance > 0 ) { DistanceTargetPitchOffset = 1 } else { DistanceTargetPitchOffset = 0 }

holoCreate(5, Haxis:pos(), vec(0.3), Self:toWorld(ang(0, TurretAng:yaw(), 0)))

Haxis:parentTo(holoEntity(5))
Turret:parentTo(holoEntity(6))

GunDir = (Base:pos() - Target:pos())

TurretAng = angnorm(clamp(TurretAng - clamp(heading(vec(), TurretAng, Self:toLocalAxis(GunDir)), -360, 360), ang(-40,-360, 0), ang(40, 360, 0)))

HaxisAngle = Self:toWorld(ang(0,clamp(TurretAng:yaw()+90,-360,360),0))
holoAng(5, HaxisAngle)

holoAng(6, holoEntity(5):toWorld(VaxisAngle))

LOSfilter = Turret:getConnectedEntities()
LOSfilter:pushEntity(Turret)

rangerFilter(LOSfilter)
LOSranger =  rangerOffsetHull(20000, Turret:pos(), Turret:forward(), vec(0.5))

holoCreate(7)
holoPos(7, LOSranger:position())
holoScale(7, vec(0.5))

        }

 else {

    FilterArray = entity():getConnectedEntities()
    FilterArray:pushEntity(entity())

    local Network = Networks[1, table]
    local NetworkInputs = array()

    for(R = 0, Rays) {

    if(ShowRaycastHolos == 1) {

    holoCreate(R+100000)
    holoColor(R+100000, vec(random(255), random(255), random(255)))
    holoScale(R+100000, vec(1))
    holoPos(R+100000, RangerFront:pos())

        }

    rangerFilter(FilterArray)

    RangerFront = rangerOffset(RayDist, Base:pos(), vec(sin(Delta * R), cos(Delta * R), 0):rotate(Base:forward():toAngle()-ang(0,180,0)))
    
    FRONT = RangerFront:distance()

    NetworkInputs:pushNumber(FRONT)

    }

    Network:nSetInputs(table("inputs" = NetworkInputs))

    local NetworkResult = Network:nGetResult()

    if(NetworkInputs:min() > 325) {
        A = 0 TURN = 0
        Gear = 1
        Throttle = 75
    } 

    elseif(TURN == 0 & HUNT == 1 & sign(HuntY) == 1) { A = 1, D = 0 }
    elseif(TURN == 0 & HUNT == 1 & sign(HuntY) == -1) { A = 0, D = 1 }
    
    elseif(NetworkInputs:min() < 100) {
    Gear = 2
    }
    
    else {
    A = clamp(NetworkResult[2, number], -1, 1) 
    TURN = 1
    Throttle = 50
    }

}
